Phần B – Đoán output

B1

class Player {
    std::string name;
public:
    Player(std::string n) : name{n} { cout << "C:" << name << " "; }
    ~Player() { cout << "D:" << name << " "; }
};
int main() {
    Player a {"A"};
    { Player b {"B"}; }
    Player c {"C"};
}
Đáp ánC:A C:B D:B C:C D:C D:A — b bị hủy khi hết khối; c và a bị hủy theo thứ tự ngược.

B2

class P {
public:
    P(int x) { cout << "int "; }
    P() : P{0} { cout << "none "; }
    P(const P &s) : P{} { cout << "copy "; }
};
void f(P p) {}
int main() { P a; f(a); }
Đáp ánint none int none copy — P a → int, none. f(a) sao chép → ủy quyền P{} → int, none, rồi copy.

    static int count;
public:
    Player() { ++count; }
    ~Player() { --count; }
    static int get() { return count; }
};
int Player::count {0};

    Player a, b;

Gợi ý – Bài 4 deep copy
IntArray(const IntArray &src) : size{src.size}, data{new int[src.size]} {
    for (size_t i = 0; i < size; ++i) data[i] = src.data[i];
}
~IntArray() { delete [] data; }
