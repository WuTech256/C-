Part B – Predict the output

B1

class Player {
    std::string name;
public:
    Player(std::string n) : name{n} { cout << "C:" << name << " "; }
    ~Player() { cout << "D:" << name << " "; }
};
int main() {
    Player a {"A"};
    { Player b {"B"}; }
    Player c {"C"};
}
AnswerC:A C:B D:B C:C D:C D:A — b dies at the end of its block; c and a are destroyed in reverse order.

B2

class P {
public:
    P(int x) { cout << "int "; }
    P() : P{0} { cout << "none "; }
    P(const P &s) : P{} { cout << "copy "; }
};
void f(P p) {}
int main() { P a; f(a); }
Answerint none int none copy — P a → int, none. f(a) copies → delegates to P{} → int, none, then copy.
